using UnityEngine;

public class SimpleMigrationTool : MonoBehaviour
{
    [Header("One-Click Migration")]
    public bool showDebugInfo = true;
    
    void Start()
    {
        if (showDebugInfo)
        {
            Debug.Log("=== Simple Migration Tool Ready ===");
            Debug.Log("Use the buttons in the Inspector or the on-screen UI to migrate your objects");
        }
    }
    
    [ContextMenu("Migrate All Objects")]
    public void MigrateAllObjects()
    {
        Debug.Log("=== STARTING MIGRATION ===");
        
        int migratedCount = 0;
        
        // Find all objects with old PushableObjects component
        PushableObjects[] oldPushables = FindObjectsByType<PushableObjects>(FindObjectsSortMode.None);
        
        Debug.Log($"Found {oldPushables.Length} objects with old PushableObjects to migrate");
        
        foreach (PushableObjects oldPushable in oldPushables)
        {
            GameObject obj = oldPushable.gameObject;
            
            Debug.Log($"Migrating: {obj.name}");
            
            // Remove old components
            DestroyImmediate(oldPushable);
            
            // Remove MultiTag if it exists
            MultiTag multiTag = obj.GetComponent<MultiTag>();
            if (multiTag) 
            {
                DestroyImmediate(multiTag);
            }
            
            // Set to pushable layer (layer 6)
            obj.layer = 6; // Pushable layer
            
            // Make sure it has Rigidbody
            if (obj.GetComponent<Rigidbody>() == null)
            {
                Rigidbody rb = obj.AddComponent<Rigidbody>();
                rb.isKinematic = true;
                rb.useGravity = false;
            }
            
            // Make sure it has a collider
            if (obj.GetComponent<Collider>() == null)
            {
                obj.AddComponent<BoxCollider>();
            }
            
            // Snap to grid (round position to nearest integer)
            Vector3 pos = obj.transform.position;
            obj.transform.position = new Vector3(
                Mathf.Round(pos.x),
                Mathf.Round(pos.y), 
                Mathf.Round(pos.z)
            );
            
            migratedCount++;
            Debug.Log($"✓ Successfully migrated: {obj.name}");
        }
        
        Debug.Log($"=== MIGRATION COMPLETE! ===");
        Debug.Log($"Successfully migrated {migratedCount} objects");
        Debug.Log("Objects are now on the Pushable layer and grid-aligned");
        Debug.Log("You can now use your new clean PlayerController system!");
    }
    
    [ContextMenu("Check Status")]
    public void CheckMigrationStatus()
    {
        PushableObjects[] oldObjects = FindObjectsByType<PushableObjects>(FindObjectsSortMode.None);
        
        Debug.Log("=== MIGRATION STATUS ===");
        Debug.Log($"Objects with old PushableObjects component: {oldObjects.Length}");
        
        if (oldObjects.Length == 0)
        {
            Debug.Log("✅ MIGRATION COMPLETE! All objects ready for new system.");
        }
        else
        {
            Debug.Log("⚠️ Migration needed. Use 'Migrate All Objects' button.");
            foreach (var obj in oldObjects)
            {
                Debug.Log($"  - {obj.name} (needs migration)");
            }
        }
    }
    
    [ContextMenu("Setup Core Systems")]
    public void SetupCoreSystems()
    {
        Debug.Log("Setting up core game systems...");
        
        // Add InputHandler if missing
        MountPuzzle.Core.InputHandler inputHandler = FindFirstObjectByType<MountPuzzle.Core.InputHandler>();
        if (inputHandler == null)
        {
            GameObject inputObj = new GameObject("Input Handler");
            inputObj.AddComponent<MountPuzzle.Core.InputHandler>();
            Debug.Log("✓ Created Input Handler");
        }
        else
        {
            Debug.Log("✓ Input Handler already exists");
        }
    }
    
    void OnGUI()
    {
        if (!showDebugInfo) return;
        
        // Create UI buttons for easy access
        GUIStyle buttonStyle = new GUIStyle(GUI.skin.button);
        buttonStyle.fontSize = 14;
        buttonStyle.fontStyle = FontStyle.Bold;
        
        GUIStyle boxStyle = new GUIStyle(GUI.skin.box);
        boxStyle.fontSize = 12;
        
        // Main panel
        GUI.Box(new Rect(20, 20, 250, 120), "Migration Tool", boxStyle);
        
        if (GUI.Button(new Rect(30, 50, 200, 30), "MIGRATE ALL OBJECTS", buttonStyle))
        {
            MigrateAllObjects();
        }
        
        if (GUI.Button(new Rect(30, 85, 95, 25), "Check Status", GUI.skin.button))
        {
            CheckMigrationStatus();
        }
        
        if (GUI.Button(new Rect(135, 85, 95, 25), "Setup Core", GUI.skin.button))
        {
            SetupCoreSystems();
        }
        
        if (GUI.Button(new Rect(30, 115, 80, 20), "Hide Panel", GUI.skin.button))
        {
            showDebugInfo = false;
        }
    }
}
