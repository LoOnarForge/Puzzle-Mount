using UnityEngine;
using MountPuzzle.Utilities;

namespace MountPuzzle.Puzzle
{
    [RequireComponent(typeof(Rigidbody))]
    public class PushableBlock : MonoBehaviour
    {
        [Header("Movement Settings")]
        [SerializeField] private float moveSpeed = 2f;
        [SerializeField] private float pushCooldown = 0.1f;
        
        [Header("Visual Feedback")]
        [SerializeField] private Color normalColor = Color.white;
        [SerializeField] private Color highlightColor = Color.yellow;
        [SerializeField] private float colorTransitionSpeed = 5f;
        
        [Header("Physics Settings")]
        [SerializeField] private LayerMask groundLayers = 1 << 5;
        [SerializeField] private LayerMask obstacleLayers = (1 << 5) | (1 << 6); // Ground + Pushable
        
        private Rigidbody rb;
        private Renderer blockRenderer;
        private Material blockMaterial;
        
        private Vector3 targetPosition;
        private bool isMoving;
        private bool isHighlighted;
        private Color currentColor;
        private float lastPushTime;
        
        public bool IsMoving => isMoving;
        public bool IsHighlighted => isHighlighted;
        
        private void Awake()
        {
            rb = GetComponent<Rigidbody>();
            blockRenderer = GetComponent<Renderer>();
            
            SetupRigidbody();
            SetupMaterial();
            InitializePosition();
        }
        
        private void SetupRigidbody()
        {
            rb.isKinematic = true;
            rb.useGravity = false;
        }
        
        private void SetupMaterial()
        {
            if (blockRenderer && blockRenderer.material)
            {
                // Create a unique material instance to avoid affecting other blocks
                blockMaterial = new Material(blockRenderer.material);
                blockRenderer.material = blockMaterial;
                
                normalColor = blockMaterial.color;
                currentColor = normalColor;
            }
        }
        
        private void InitializePosition()
        {
            targetPosition = GridSnapper.SnapToGrid(transform.position);
            transform.position = targetPosition;
        }
        
        private void Update()
        {
            HandleMovement();
            UpdateVisuals();
            CheckForFalling();
        }
        
        private void HandleMovement()
        {
            if (!isMoving) return;
            
            transform.position = Vector3.MoveTowards(
                transform.position, 
                targetPosition, 
                moveSpeed * Time.deltaTime
            );
            
            if (Vector3.Distance(transform.position, targetPosition) < 0.01f)
            {
                transform.position = targetPosition;
                isMoving = false;
                
                // Snap to grid after movement
                GridSnapper.SnapObjectToGrid(transform);
            }
        }
        
        private void UpdateVisuals()
        {
            if (!blockMaterial) return;
            
            Color targetColor = isHighlighted ? highlightColor : normalColor;
            currentColor = Color.Lerp(currentColor, targetColor, colorTransitionSpeed * Time.deltaTime);
            blockMaterial.color = currentColor;
        }
        
        private void CheckForFalling()
        {
            if (isMoving) return;
            
            // Check if block is supported by ground
            if (!Physics.Raycast(transform.position, Vector3.down, 1.1f, groundLayers))
            {
                StartFalling();
            }
        }
        
        public bool TryPush(Vector3 direction)
        {
            if (!CanBePushed(direction))
                return false;
            
            Vector3 pushDirection = GetCardinalDirection(direction);
            Vector3 newPosition = transform.position + pushDirection;
            
            if (!IsPositionFree(newPosition))
                return false;
            
            StartMoving(newPosition);
            lastPushTime = Time.time;
            
            return true;
        }
        
        private bool CanBePushed(Vector3 direction)
        {
            if (isMoving) return false;
            if (Time.time - lastPushTime < pushCooldown) return false;
            if (direction.magnitude < 0.1f) return false;
            
            return true;
        }
        
        private Vector3 GetCardinalDirection(Vector3 input)
        {
            Vector3 direction = Vector3.zero;
            
            if (Mathf.Abs(input.x) > Mathf.Abs(input.z))
            {
                direction.x = Mathf.Sign(input.x);
            }
            else
            {
                direction.z = Mathf.Sign(input.z);
            }
            
            return direction;
        }
        
        private bool IsPositionFree(Vector3 position)
        {
            Collider[] obstacles = Physics.OverlapBox(
                position, 
                Vector3.one * 0.4f, 
                Quaternion.identity, 
                obstacleLayers
            );
            
            // Filter out self
            foreach (Collider obstacle in obstacles)
            {
                if (obstacle.gameObject != gameObject)
                    return false;
            }
            
            return true;
        }
        
        private void StartMoving(Vector3 newPosition)
        {
            targetPosition = newPosition;
            isMoving = true;
        }
        
        private void StartFalling()
        {
            rb.isKinematic = false;
            rb.useGravity = true;
            isMoving = false;
        }
        
        private void OnCollisionEnter(Collision collision)
        {
            // Stop falling when hitting ground
            if (((1 << collision.gameObject.layer) & groundLayers) != 0)
            {
                StopFalling();
            }
        }
        
        private void StopFalling()
        {
            rb.isKinematic = true;
            rb.useGravity = false;
            rb.linearVelocity = Vector3.zero;
            
            // Snap to grid after falling
            GridSnapper.SnapObjectToGrid(transform);
            targetPosition = transform.position;
        }
        
        public void SetHighlighted(bool highlighted)
        {
            isHighlighted = highlighted;
        }
        
        private void OnDrawGizmosSelected()
        {
            Gizmos.color = isMoving ? Color.red : Color.green;
            Gizmos.DrawWireCube(transform.position, Vector3.one);
            
            if (isMoving)
            {
                Gizmos.color = Color.blue;
                Gizmos.DrawWireCube(targetPosition, Vector3.one * 1.1f);
                Gizmos.DrawLine(transform.position, targetPosition);
            }
        }
        
        private void OnDestroy()
        {
            if (blockMaterial)
            {
                DestroyImmediate(blockMaterial);
            }
        }
    }
}
