using UnityEngine;
using MountPuzzle.Player;
using MountPuzzle.Camera;
using MountPuzzle.Puzzle;
using MountPuzzle.Utilities;

namespace MountPuzzle.Core
{
    public class SetupHelper : MonoBehaviour
    {
        [Header("Auto Setup Options")]
        [SerializeField] private bool setupOnStart = true;
        [SerializeField] private bool showSetupInstructions = true;
        
        [Header("Setup References")]
        [SerializeField] private GameObject playerPrefab;
        [SerializeField] private GameObject cameraPrefab;
        
        private void Start()
        {
            if (setupOnStart)
            {
                CheckAndSetupScene();
            }
        }
        
        [ContextMenu("Setup Scene")]
        public void CheckAndSetupScene()
        {
            SetupGameManager();
            SetupInputHandler();
            SetupPlayer();
            SetupCamera();
            SetupPuzzleManager();
            AutoMigrateOldComponents();
            
            Debug.Log("Scene setup complete! Check the console for any warnings.");
        }
        
        private void AutoMigrateOldComponents()
        {
            Debug.Log("Checking for legacy components to migrate...");
            AutoMigration.CheckAndAutoMigrate();
        }
        
        private void SetupGameManager()
        {
            if (FindFirstObjectByType<GameManager>() == null)
            {
                GameObject gameManagerObj = new GameObject("Game Manager");
                gameManagerObj.AddComponent<GameManager>();
                Debug.Log("✓ Created Game Manager");
            }
            else
            {
                Debug.Log("✓ Game Manager already exists");
            }
        }
        
        private void SetupInputHandler()
        {
            if (FindFirstObjectByType<InputHandler>() == null)
            {
                GameObject inputObj = new GameObject("Input Handler");
                inputObj.AddComponent<InputHandler>();
                Debug.Log("✓ Created Input Handler");
            }
            else
            {
                Debug.Log("✓ Input Handler already exists");
            }
        }
        
        private void SetupPlayer()
        {
            PlayerController player = FindFirstObjectByType<PlayerController>();
            if (player == null)
            {
                Debug.LogWarning("⚠ No PlayerController found! Please:");
                Debug.LogWarning("  1. Select your player character GameObject");
                Debug.LogWarning("  2. Add PlayerController component");
                Debug.LogWarning("  3. Add PlayerAnimator component");
                Debug.LogWarning("  4. Add CharacterController component");
            }
            else
            {
                Debug.Log("✓ Player Controller found");
                
                // Check required components
                if (!player.GetComponent<CharacterController>())
                    Debug.LogWarning("⚠ Player needs CharacterController component");
                    
                if (!player.GetComponent<PlayerAnimator>())
                    Debug.LogWarning("⚠ Player needs PlayerAnimator component");
            }
        }
        
        private void SetupCamera()
        {
            CameraController cameraController = FindFirstObjectByType<CameraController>();
            if (cameraController == null)
            {
                UnityEngine.Camera mainCamera = UnityEngine.Camera.main;
                if (mainCamera != null)
                {
                    mainCamera.gameObject.AddComponent<CameraController>();
                    Debug.Log("✓ Added CameraController to Main Camera");
                }
                else
                {
                    Debug.LogWarning("⚠ No Main Camera found! Please add CameraController to your camera");
                }
            }
            else
            {
                Debug.Log("✓ Camera Controller already exists");
            }
        }
        
        private void SetupPuzzleManager()
        {
            if (FindFirstObjectByType<PuzzleManager>() == null)
            {
                GameObject puzzleManagerObj = new GameObject("Puzzle Manager");
                puzzleManagerObj.AddComponent<PuzzleManager>();
                Debug.Log("✓ Created Puzzle Manager");
            }
            else
            {
                Debug.Log("✓ Puzzle Manager already exists");
            }
        }
        
        [ContextMenu("Setup Pushable Blocks")]
        public void SetupPushableBlocks()
        {
            GameObject[] allObjects = FindObjectsByType<GameObject>(FindObjectsSortMode.None);
            int setupCount = 0;
            
            foreach (GameObject obj in allObjects)
            {
                // Look for objects that should be pushable blocks
                if (obj.name.ToLower().Contains("cube") || 
                    obj.name.ToLower().Contains("block") ||
                    obj.name.ToLower().Contains("pushable"))
                {
                    if (obj.GetComponent<PushableBlock>() == null)
                    {
                        // Add required components
                        if (obj.GetComponent<Rigidbody>() == null)
                            obj.AddComponent<Rigidbody>();
                            
                        if (obj.GetComponent<Collider>() == null)
                            obj.AddComponent<BoxCollider>();
                            
                        obj.AddComponent<PushableBlock>();
                        
                        // Set to pushable layer
                        LayerHelper.SetObjectToPushableLayer(obj);
                        
                        // Snap to grid
                        GridSnapper.SnapObjectToGrid(obj.transform);
                        
                        setupCount++;
                        Debug.Log($"✓ Setup pushable block: {obj.name}");
                    }
                }
            }
            
            Debug.Log($"Setup complete! Configured {setupCount} pushable blocks");
        }
        
        [ContextMenu("Show Help")]
        public void ShowSetupInstructions()
        {
            Debug.Log("=== MOUNT PUZZLE SETUP GUIDE ===");
            Debug.Log("1. Click 'Setup Scene' button to auto-setup core systems");
            Debug.Log("2. Select your player character and add:");
            Debug.Log("   - PlayerController component");
            Debug.Log("   - PlayerAnimator component"); 
            Debug.Log("   - CharacterController component");
            Debug.Log("3. Click 'Setup Pushable Blocks' to auto-setup pushable objects");
            Debug.Log("4. Configure your Animator Controller with parameters:");
            Debug.Log("   - Speed (float), IsPushing (bool), IsGrounded (bool), IsFalling (bool)");
            Debug.Log("5. Make sure your layers are set:");
            Debug.Log("   - Layer 5: 'Ground'");
            Debug.Log("   - Layer 6: 'Pushable'");
            Debug.Log("=== Controls: WASD=Move, Shift=Run, Space=Push ===");
        }
        
        private void OnGUI()
        {
            if (!showSetupInstructions) return;
            
            GUIStyle boxStyle = new GUIStyle(GUI.skin.box);
            boxStyle.fontSize = 12;
            boxStyle.normal.textColor = Color.white;
            
            GUIStyle buttonStyle = new GUIStyle(GUI.skin.button);
            buttonStyle.fontSize = 12;
            
            GUI.Box(new Rect(Screen.width - 320, 10, 300, 120), "", boxStyle);
            
            GUI.Label(new Rect(Screen.width - 310, 20, 280, 20), "Mount Puzzle - Clean Setup", boxStyle);
            
            if (GUI.Button(new Rect(Screen.width - 310, 45, 130, 25), "Setup Scene", buttonStyle))
            {
                CheckAndSetupScene();
            }
            
            if (GUI.Button(new Rect(Screen.width - 170, 45, 150, 25), "Setup Pushable Blocks", buttonStyle))
            {
                SetupPushableBlocks();
            }
            
            if (GUI.Button(new Rect(Screen.width - 310, 75, 100, 25), "Show Help", buttonStyle))
            {
                ShowSetupInstructions();
            }
            
            if (GUI.Button(new Rect(Screen.width - 200, 75, 90, 25), "Hide Panel", buttonStyle))
            {
                showSetupInstructions = false;
            }
            
            GUI.Label(new Rect(Screen.width - 310, 105, 280, 20), "Check Console for detailed instructions", boxStyle);
        }
    }
}
