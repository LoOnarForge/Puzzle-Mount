using UnityEngine;

namespace MountPuzzle.Utilities
{
    public static class GridSnapper
    {
        public static float GridSize = 1f;
        
        public static Vector3 SnapToGrid(Vector3 position)
        {
            return new Vector3(
                Mathf.Round(position.x / GridSize) * GridSize,
                Mathf.Round(position.y / GridSize) * GridSize,
                Mathf.Round(position.z / GridSize) * GridSize
            );
        }
        
        public static Vector3 SnapToGridXZ(Vector3 position)
        {
            return new Vector3(
                Mathf.Round(position.x / GridSize) * GridSize,
                position.y,
                Mathf.Round(position.z / GridSize) * GridSize
            );
        }
        
        public static void SnapObjectToGrid(Transform objectTransform)
        {
            objectTransform.position = SnapToGrid(objectTransform.position);
        }
        
        public static void SnapObjectToGridXZ(Transform objectTransform)
        {
            objectTransform.position = SnapToGridXZ(objectTransform.position);
        }
        
        public static Vector3Int GetGridCoordinates(Vector3 worldPosition)
        {
            return new Vector3Int(
                Mathf.RoundToInt(worldPosition.x / GridSize),
                Mathf.RoundToInt(worldPosition.y / GridSize),
                Mathf.RoundToInt(worldPosition.z / GridSize)
            );
        }
        
        public static Vector3 GridCoordinatesToWorld(Vector3Int gridCoords)
        {
            return new Vector3(
                gridCoords.x * GridSize,
                gridCoords.y * GridSize,
                gridCoords.z * GridSize
            );
        }
        
        public static bool IsOnGrid(Vector3 position, float tolerance = 0.01f)
        {
            Vector3 snapped = SnapToGrid(position);
            return Vector3.Distance(position, snapped) <= tolerance;
        }
    }
}
