using UnityEngine;

namespace MountPuzzle.Input
{
    public class SimpleInputHandler : MonoBehaviour
    {
        [Header("Input Settings")]
        [SerializeField] private float inputDeadzone = 0.1f;
        [SerializeField] private KeyCode runKey = KeyCode.LeftShift;
        [SerializeField] private KeyCode pushKey = KeyCode.Space;
        
        private Vector2 moveInput;
        private bool runInput;
        private bool pushInput;
        
        public Vector2 MoveInput => moveInput;
        public bool IsRunning => runInput;
        public bool IsPushing => pushInput;
        public bool HasMoveInput => moveInput.magnitude > inputDeadzone;
        
        private void Update()
        {
            HandleKeyboardInput();
        }
        
        private void HandleKeyboardInput()
        {
            // Movement input
            float horizontal = 0f;
            float vertical = 0f;
            
            if (UnityEngine.Input.GetKey(KeyCode.W) || UnityEngine.Input.GetKey(KeyCode.UpArrow))
                vertical += 1f;
            if (UnityEngine.Input.GetKey(KeyCode.S) || UnityEngine.Input.GetKey(KeyCode.DownArrow))
                vertical -= 1f;
            if (UnityEngine.Input.GetKey(KeyCode.A) || UnityEngine.Input.GetKey(KeyCode.LeftArrow))
                horizontal -= 1f;
            if (UnityEngine.Input.GetKey(KeyCode.D) || UnityEngine.Input.GetKey(KeyCode.RightArrow))
                horizontal += 1f;
            
            moveInput = new Vector2(horizontal, vertical);
            
            // Apply deadzone
            if (moveInput.magnitude < inputDeadzone)
                moveInput = Vector2.zero;
            else
                moveInput = moveInput.normalized;
            
            // Run input
            runInput = UnityEngine.Input.GetKey(runKey);
            
            // Push input
            pushInput = UnityEngine.Input.GetKey(pushKey);
        }
        
        public Vector3 GetWorldMoveDirection()
        {
            return new Vector3(moveInput.x, 0f, moveInput.y).normalized;
        }
        
        private void OnGUI()
        {
            if (!MountPuzzle.Core.GameManager.Instance || !MountPuzzle.Core.GameManager.Instance.debugMode) return;
            
            GUIStyle style = new GUIStyle();
            style.fontSize = 14;
            style.normal.textColor = Color.yellow;
            
            GUI.Label(new Rect(10, 50, 300, 20), $"Move: ({moveInput.x:F2}, {moveInput.y:F2})", style);
            GUI.Label(new Rect(10, 70, 300, 20), $"Run: {runInput} | Push: {pushInput}", style);
            GUI.Label(new Rect(10, 90, 300, 20), $"Controls: WASD/Arrows=Move, {runKey}=Run, {pushKey}=Push", style);
        }
    }
}
