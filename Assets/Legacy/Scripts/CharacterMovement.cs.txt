

using UnityEngine;
using UnityEngine.InputSystem;
using System.Collections;
using System.Collections.Generic;
using System;
public class CharacterMovement : MonoBehaviour
{

    public float normalSpeed = 5f;
    public float pushingSpeed = 2f;
    public float runningSpeed = 10f;

    [Header("Ground Detection Settings")]
    public float detectionRadius = 0.4f;
    public LayerMask groundLayer;

    [Header("Falling Settings")]
    public float gravity = 9.81f;
    public float fallMultiplier = 2f;
    public float coyoteTime = 0.2f;
    public float fallAnimationDelay = 0.1f;
    public float fallMomentumDecay = 2f;

    [Header("Pushing Settings")]
    public float pushBreakDelay = 0.2f;
    public float pushDetectionDistance = 1f;
    public float pushHoldTime = 0.5f;
    public LayerMask pushableLayer;
    private float postPushCooldownTimer = 0f;
    private bool hasPushedOnce = false;
    
    [Header("Selection Settings")]
    private float secondaryBoxcastExtension = 0.3f;
    private Coroutine deselectionCoroutine = null;
    private CharacterController controller;
    private Animator animator;
    private float rotationSpeed = 600f;
    private Vector2 moveInput;
    private float currentSpeed;
    private float verticalVelocity;
    private bool isGrounded;
    private bool isFalling;
    private float coyoteTimer;
    private float fallAnimationTimer;
    private Vector3 fallMomentum;
    public bool isPushing = false;
    public GameObject currentPushableObject;
    private GameObject lastPushableObject;
    public Vector3 lockedPushDirection { get; private set; }
    private float pushTimer = 0f;
    private bool isRotating = false;
    private CubeSelection cubeSelection;
    private PlayerControls playerControls;

    private void Awake()
    {
        controller = GetComponent<CharacterController>();
        animator = GetComponent<Animator>();
        cubeSelection = GetComponent<CubeSelection>();
        playerControls = new PlayerControls();

        if (cubeSelection == null)
        {
            Debug.LogError("CubeSelection script is missing on the Player!");
        }
    }
    private void Update()
    {
        CheckGrounded();
        HandleMovement();
        CheckForPushable();
        HandleAnimation();
        HandleRotationInput();
        DrawPrimaryRaycast();
    }


    public void CheckForPushable()
    {
        if (isRotating && currentPushableObject != null) return;

        GameObject bestCandidate = FindBestPushableCandidate();

        if (bestCandidate == null && currentPushableObject != null)
        {
            Vector3 origin = new Vector3(controller.bounds.center.x, controller.bounds.min.y + 0.8f, controller.bounds.center.z);
            Vector3 direction = transform.forward;
            float extendedDistance = pushDetectionDistance + secondaryBoxcastExtension;

            if (Physics.Raycast(origin, direction, out RaycastHit hit, extendedDistance, pushableLayer))
            {
                if (hit.collider.gameObject == currentPushableObject)
                {
                    bestCandidate = currentPushableObject;
                }
            }
        }

        UpdatePushableSelection(bestCandidate);
        HandleDeselectionWhenNoCandidate(bestCandidate);
        HandlePushingLogic(bestCandidate);
    }
    private GameObject FindBestPushableCandidate()
    {
        Dictionary<GameObject, int> cubeScores = new Dictionary<GameObject, int>();
        Vector3 origin = new Vector3(controller.bounds.center.x, controller.bounds.min.y + 0.3f, controller.bounds.center.z);
        float[] angles = new float[] { 0f, -5f, 5f, -15f, 15f, -30f, 30f };
        int[] weights = new int[] { 3, 2, 2, 1, 1, 1, 1 };

        for (int i = 0; i < angles.Length; i++)
        {
            Vector3 dir = Quaternion.Euler(0, angles[i], 0) * transform.forward;
            if (Physics.Raycast(origin, dir, out RaycastHit hit, pushDetectionDistance, pushableLayer))
            {
                GameObject cube = hit.collider.gameObject;
                if (cubeScores.ContainsKey(cube))
                    cubeScores[cube] += weights[i];
                else
                    cubeScores[cube] = weights[i];
            }
        }

        if (cubeScores.Count == 0) return null;

        GameObject best = null;
        int bestScore = 0;

        foreach (var kvp in cubeScores)
        {
            int threshold = cubeScores.Count == 1 ? 5 : 7;

            if (kvp.Value >= threshold && kvp.Value > bestScore)
            {
                best = kvp.Key;
                bestScore = kvp.Value;
            }
        }

        return best;
    }
    private void UpdatePushableSelection(GameObject bestCandidate)
    {
        if (bestCandidate != null && currentPushableObject != bestCandidate)
        {
            if (lastPushableObject != null)
            {
                PushableObjects lastCube = lastPushableObject.GetComponent<PushableObjects>();
                if (lastCube != null) lastCube.SetSelected(false);
            }

            if (deselectionCoroutine != null)
            {
                StopCoroutine(deselectionCoroutine);
                deselectionCoroutine = null;
            }

            currentPushableObject = bestCandidate;

            PushableObjects currentCube = currentPushableObject.GetComponent<PushableObjects>();
            if (currentCube != null)
            {
                lockedPushDirection = currentCube.GetRelativePushDirection(transform);
                pushTimer = 0f;
                hasPushedOnce = false;

                if (cubeSelection != null)
                    cubeSelection.SelectBottomThreeCubes(currentPushableObject);

                lastPushableObject = bestCandidate;
            }
        }
    }
    private void HandleDeselectionWhenNoCandidate(GameObject bestCandidate)
    {
        if (bestCandidate == null && currentPushableObject != null)
        {
            if (deselectionCoroutine == null && cubeSelection != null)
            {
                deselectionCoroutine = StartCoroutine(cubeSelection.WaitBeforeDeselecting(currentPushableObject));
            }
        }
        else
        {
            if (deselectionCoroutine != null)
            {
                StopCoroutine(deselectionCoroutine);
                deselectionCoroutine = null;
            }
        }
    }
    private void HandlePushingLogic(GameObject bestCandidate)
    {
        if (postPushCooldownTimer > 0f)
        {
            postPushCooldownTimer -= Time.deltaTime;
            return;
        }

        if (bestCandidate != null && moveInput != Vector2.zero)
        {
            PushableObjects cube = bestCandidate.GetComponent<PushableObjects>();

            if (!isPushing)
            {
                isPushing = true;
                animator.SetBool("isPushing", true);
            }

            float currentHoldThreshold = hasPushedOnce ? pushHoldTime * 0.5f : pushHoldTime;
            pushTimer += Time.deltaTime;

            if (cube != null && !cube.IsMoving())
            {
                List<PushableObjects> stack = GetStackOfPushables(cube);

                if (stack.Count > 3)
                    return;

                if (pushTimer >= currentHoldThreshold)
                {
                    bool allCubesSupported = stack.TrueForAll(c =>
                        Physics.Raycast(c.transform.position, Vector3.down, 1f, LayerMask.GetMask("Ground", "Pushable"))
                    );

                    if (allCubesSupported && stack.TrueForAll(c => !c.IsMoving()) && IsStackPathFree(stack, lockedPushDirection))
                    {
                        foreach (var c in stack)
                            c.TryPush(lockedPushDirection);

                        postPushCooldownTimer = Keyboard.current.shiftKey.isPressed ? pushBreakDelay : 0.01f;
                        pushTimer = 0f;
                        hasPushedOnce = true;
                    }
                }
            }
        }
        else
        {
            if (isPushing)
            {
                isPushing = false;
                animator.SetBool("isPushing", false);
            }
            pushTimer = 0f;
            hasPushedOnce = false;
        }
    }

    private void CheckGrounded()
    {
        Vector3 sphereCenter = transform.position + Vector3.down * 0.1f;
        Collider[] hitColliders = Physics.OverlapSphere(sphereCenter, detectionRadius, groundLayer);

        isGrounded = hitColliders.Length > 0;

        if (isGrounded)
        {
            isFalling = false;
            coyoteTimer = coyoteTime;
            fallAnimationTimer = fallAnimationDelay;
            verticalVelocity = -2f;
            animator.SetBool("isFalling", false);
        }
        else
        {
            if (coyoteTimer > 0)
            {
                coyoteTimer -= Time.deltaTime;
            }
            else
            {
                if (!isFalling)
                {
                    isFalling = true;
                    fallMomentum = new Vector3(moveInput.x, 0, moveInput.y).normalized * currentSpeed;
                }

                fallAnimationTimer -= Time.deltaTime;
                if (fallAnimationTimer <= 0)
                {
                    animator.SetBool("isFalling", true);
                }
            }
        }
    }
    private void HandleMovement()
    {
        Vector3 moveDirection = new Vector3(moveInput.x, 0, moveInput.y).normalized;

        bool isRunning = Keyboard.current.shiftKey.isPressed && !isPushing;
        currentSpeed = moveInput.magnitude > 0.1f ? (isRunning ? runningSpeed : (isPushing ? pushingSpeed : normalSpeed)) : 0f;

        Vector3 worldMove = moveDirection * currentSpeed;

        if (isGrounded)
        {
            RaycastHit hit;
            Vector3 rayOrigin = transform.position + Vector3.up * 0.1f;

            if (Physics.Raycast(rayOrigin, Vector3.down, out hit, 1f, groundLayer))
            {
                Vector3 slopeNormal = hit.normal;
                float slopeAngle = Vector3.Angle(slopeNormal, Vector3.up);

                if (slopeAngle <= 45f)
                {
                    worldMove = Vector3.ProjectOnPlane(worldMove, slopeNormal).normalized * currentSpeed;
                }
            }
        }

        if (isFalling)
        {
            verticalVelocity -= gravity * fallMultiplier * Time.deltaTime;
            fallMomentum = Vector3.Lerp(fallMomentum, Vector3.zero, fallMomentumDecay * Time.deltaTime);
            worldMove = fallMomentum;
        }

        worldMove.y = verticalVelocity;
        controller.Move(worldMove * Time.deltaTime);

        if (moveDirection != Vector3.zero && !isFalling)
        {
            Vector3 flatDirection = new Vector3(worldMove.x, 0, worldMove.z).normalized;
            Quaternion toRotation = Quaternion.LookRotation(flatDirection);
            transform.rotation = Quaternion.RotateTowards(transform.rotation, toRotation, rotationSpeed * Time.deltaTime);
        }
    }
    private void HandleAnimation()
    {
        animator.SetFloat("Speed", currentSpeed);
        animator.SetFloat("MoveDirectionX", moveInput.x);
        animator.SetFloat("MoveDirectionY", moveInput.y);
        animator.SetBool("isGrounded", isGrounded);
        animator.SetBool("isFalling", isFalling);
        animator.SetBool("isPushing", isPushing);
    }

    public void OnMove(InputAction.CallbackContext context)
    {
        moveInput = context.ReadValue<Vector2>();
    }
    public bool IsPositionFree(Vector3 position)
    {
        Collider[] hits = Physics.OverlapBox(position, new Vector3(0.45f, 0.45f, 0.45f), Quaternion.identity, LayerMask.GetMask("Ground"));
        return hits.Length == 0;
    }
    public void HandleRotationInput()
    {
        if (cubeSelection == null || isRotating || moveInput.magnitude > 0.1f) return;

        GameObject activeCube = cubeSelection.GetActiveCube();
        if (activeCube == null) return;

        if (Keyboard.current.qKey.wasPressedThisFrame)
        {
            StartCoroutine(RotateCubeCoroutine(activeCube, 90f, Vector3.up));
        }

        if (Keyboard.current.eKey.wasPressedThisFrame)
        {
            StartCoroutine(RotateCubeCoroutine(activeCube, 90f, Vector3.right));
        }
    }
    
    private IEnumerator RotateCubeCoroutine(GameObject cube, float angle, Vector3 axis)
    {
        if (cube == null || isRotating) yield break;

        isRotating = true;

        Quaternion startRotation = cube.transform.rotation;
        Quaternion endRotation = Quaternion.AngleAxis(angle, axis) * startRotation;
        float rotationProgress = 0f;
        float rotationSpeed = 5f;

        while (rotationProgress < 1f)
        {
            rotationProgress += Time.deltaTime * rotationSpeed;
            cube.transform.rotation = Quaternion.Slerp(startRotation, endRotation, rotationProgress);
            yield return null;
        }

        cube.transform.rotation = endRotation;

        isRotating = false;
    }
    public void OnChangeCubeSelection(InputAction.CallbackContext context)
    {
        if (context.performed && cubeSelection != null)
        {
            cubeSelection.SwitchActiveCube();
        }
    }


    private List<PushableObjects> GetStackOfPushables(PushableObjects bottomCube)
    {
        List<PushableObjects> stack = new List<PushableObjects> { bottomCube };
        Vector3 checkPosition = bottomCube.transform.position + new Vector3(0, 0.95f, 0);

        while (true)
        {
            Collider[] hits = Physics.OverlapBox(checkPosition, Vector3.one * 0.4f, Quaternion.identity, pushableLayer);
            bool foundNext = false;

            foreach (var hit in hits)
            {
                PushableObjects cube = hit.GetComponent<PushableObjects>();
                if (cube != null && !stack.Contains(cube))
                {
                    stack.Add(cube);
                    checkPosition += new Vector3(0, 0.95f, 0);
                    foundNext = true;
                    break;
                }
            }

            if (!foundNext)
                break;
        }

        return stack;
    }

    private bool IsStackPathFree(List<PushableObjects> cubes, Vector3 direction)
    {
        foreach (var cube in cubes)
        {
            Vector3 targetPosition = cube.transform.position + direction;
            Collider[] hits = Physics.OverlapBox(targetPosition, new Vector3(0.45f, 0.45f, 0.45f), Quaternion.identity, LayerMask.GetMask("Pushable", "Ground"));

            foreach (var hit in hits)
            {
                if (!cubes.Contains(hit.GetComponent<PushableObjects>()))
                {
                    return false;
                }
            }
        }

        return true;
    }



    private void OnEnable()
    {
        playerControls.PlayerActions.ChangeCubeSelection.performed += OnChangeCubeSelection;
        playerControls.PlayerActions.Enable();
    }
    private void OnDisable()
    {
        playerControls.PlayerActions.ChangeCubeSelection.performed -= OnChangeCubeSelection;
        playerControls.PlayerActions.Disable();
    }

    private void DrawPrimaryRaycast()
    {
        Vector3 origin = new Vector3(controller.bounds.center.x, controller.bounds.min.y + 0.3f, controller.bounds.center.z);
        float[] angles = { -45f, -30f, -5f, 0f, 5f, 30f, 45f };
        Color[] colors = { Color.red, Color.white, Color.yellow, Color.green, Color.yellow, Color.white, Color.red };

        for (int i = 0; i < angles.Length; i++)
        {
            Vector3 dir = Quaternion.Euler(0, angles[i], 0) * transform.forward;
            Debug.DrawRay(origin, dir * pushDetectionDistance, colors[i]);
        }
    }
}
