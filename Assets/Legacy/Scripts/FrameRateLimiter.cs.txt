using UnityEngine;

public class FrameRateLimiter : MonoBehaviour
{
    // Target frame rate, set to 60 FPS
    public int targetFrameRate = 60;

    private void Start()
    {
        // Set the target frame rate to the desired value
        Application.targetFrameRate = targetFrameRate;
    }

    private void OnGUI()
    {
        // Set up a small font style for the FPS text
        GUIStyle style = new GUIStyle();
        style.fontSize = 20;  // Small font size
        style.normal.textColor = Color.white;  // White color for visibility

        // Display the FPS in the top-right corner
        GUI.Label(new Rect(Screen.width - 100, 10, 100, 30), "FPS: " + (1.0f / Time.deltaTime).ToString("F2"), style);
    }
}
