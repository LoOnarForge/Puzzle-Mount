using UnityEngine;

public class AnimationTesting : MonoBehaviour
{
    private Animator animator;
    private Rigidbody rb;

    private bool isJumping = false;
   // private bool isPullingUp = false;

    void Start()
    {
        animator = GetComponent<Animator>();
        rb = GetComponent<Rigidbody>();

        if (animator == null)
            Debug.LogError("Animator component missing!");

        if (rb == null)
            Debug.LogError("Rigidbody component missing!");

        // Initial debug at the start
        DebugInitialState();
    }

    void Update()
    {
        DebugFrameData();

        if (Input.GetKeyDown(KeyCode.E))
        {
            StartClimbSequence();
        }

        // Additional check for unexpected state changes
        if (animator != null)
        {
            AnimatorStateInfo stateInfo = animator.GetCurrentAnimatorStateInfo(0);
            Debug.Log("Current Animation State: " + stateInfo.fullPathHash + " | Normalized Time: " + stateInfo.normalizedTime);
        }
    }

    void StartClimbSequence()
    {
        if (!isJumping)
        {
            isJumping = true;
            animator.SetTrigger("JumpGrab");
            Debug.Log("JumpGrab animation started");
        }
    }

    void FixedUpdate()
    {
        if (isJumping && rb != null)
        {
            // Apply force for testing (if needed)
            rb.linearVelocity = new Vector3(0, 5f, 0);
            Debug.Log("Applying upward velocity: " + rb.linearVelocity);
        }
    }

    void DebugInitialState()
    {
        Debug.Log("=== Initial State ===");
        Debug.Log("Rigidbody Linear Velocity: " + rb.linearVelocity);
        Debug.Log("Rigidbody Linear Damping: " + rb.linearDamping);
        Debug.Log("Rigidbody Angular Damping: " + rb.angularDamping);
        Debug.Log("Rigidbody Position: " + rb.position);
        Debug.Log("Rigidbody Rotation: " + rb.rotation);
        Debug.Log("Rigidbody IsKinematic: " + rb.isKinematic);
        Debug.Log("Rigidbody UseGravity: " + rb.useGravity);
    }

    void DebugFrameData()
    {
        if (rb != null)
        {
            Debug.Log("Frame Data - Rigidbody Linear Velocity: " + rb.linearVelocity);
            Debug.Log("Frame Data - Rigidbody Position: " + rb.position);
            Debug.Log("Frame Data - Rigidbody Rotation: " + rb.rotation);
            Debug.Log("Frame Data - Rigidbody IsKinematic: " + rb.isKinematic);
            Debug.Log("Frame Data - Rigidbody UseGravity: " + rb.useGravity);
        }
    }
}
