using UnityEngine;
using UnityEngine.UIElements;

public class UIVsMouseInteraction : MonoBehaviour
{
    public UIDocument uiDocument; // Assign in the Inspector

    private void OnEnable()
    {
        // Get the root UI object
        VisualElement root = uiDocument.rootVisualElement;

        // Find buttons by name and assign click events
        Button button1 = root.Q<Button>("Action1");
        if (button1 != null)
            button1.clicked += () => Debug.Log("Button 1 Clicked!");

        Button button2 = root.Q<Button>("Action2");
        if (button2 != null)
            button2.clicked += () => Debug.Log("Button 2 Clicked!");

        Button button3 = root.Q<Button>("Action3");
        if (button3 != null)
            button3.clicked += () => Debug.Log("Button 3 Clicked!");

        Button button4 = root.Q<Button>("Action4");
        if (button4 != null)
            button4.clicked += () => Debug.Log("Button 4 Clicked!");

        Button button5 = root.Q<Button>("Action5");
        if (button5 != null)
            button5.clicked += () => Debug.Log("Button 5 Clicked!");

        Button button6 = root.Q<Button>("Action6");
        if (button6 != null)
            button6.clicked += () => Debug.Log("Button 6 Clicked!");

    }
}
