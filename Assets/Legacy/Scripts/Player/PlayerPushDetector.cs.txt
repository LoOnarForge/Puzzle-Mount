using UnityEngine;
using MountPuzzle.Puzzle;

namespace MountPuzzle.Player
{
    [System.Serializable]
    public class PlayerPushDetector
    {
        [Header("Detection Settings")]
        [SerializeField] private float detectionHeight = 1f;
        [SerializeField] private float detectionAngle = 30f;
        
        public PushableBlock FindNearestPushable(Vector3 playerPosition, Vector3 playerForward, float range, LayerMask pushableLayers)
        {
            Vector3 rayOrigin = playerPosition + Vector3.up * detectionHeight;
            
            // Simple forward raycast
            if (Physics.Raycast(rayOrigin, playerForward, out RaycastHit hit, range, pushableLayers))
            {
                PushableBlock block = hit.collider.GetComponent<PushableBlock>();
                if (block != null && !block.IsMoving)
                {
                    return block;
                }
            }
            
            // Check slightly angled rays for better detection
            float[] angles = { -detectionAngle, detectionAngle };
            
            foreach (float angle in angles)
            {
                Vector3 direction = Quaternion.AngleAxis(angle, Vector3.up) * playerForward;
                
                if (Physics.Raycast(rayOrigin, direction, out RaycastHit angleHit, range * 0.8f, pushableLayers))
                {
                    PushableBlock block = angleHit.collider.GetComponent<PushableBlock>();
                    if (block != null && !block.IsMoving)
                    {
                        return block;
                    }
                }
            }
            
            return null;
        }
        
        public bool CanPushInDirection(Vector3 blockPosition, Vector3 direction, LayerMask obstacleLayers)
        {
            Vector3 targetPosition = blockPosition + direction;
            
            // Check if target position is clear
            Collider[] obstacles = Physics.OverlapBox(
                targetPosition, 
                Vector3.one * 0.4f, 
                Quaternion.identity, 
                obstacleLayers
            );
            
            return obstacles.Length == 0;
        }
        
        public void DrawDebugGizmos(Vector3 playerPosition, Vector3 playerForward, float range, bool foundPushable)
        {
            Vector3 rayOrigin = playerPosition + Vector3.up * detectionHeight;
            
            Gizmos.color = foundPushable ? Color.yellow : Color.blue;
            Gizmos.DrawRay(rayOrigin, playerForward * range);
            
            // Draw detection angles
            Gizmos.color = Color.cyan;
            Vector3 leftDirection = Quaternion.AngleAxis(-detectionAngle, Vector3.up) * playerForward;
            Vector3 rightDirection = Quaternion.AngleAxis(detectionAngle, Vector3.up) * playerForward;
            
            Gizmos.DrawRay(rayOrigin, leftDirection * range * 0.8f);
            Gizmos.DrawRay(rayOrigin, rightDirection * range * 0.8f);
        }
    }
}
