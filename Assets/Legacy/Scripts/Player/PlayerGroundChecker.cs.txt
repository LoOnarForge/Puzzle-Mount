using UnityEngine;

namespace MountPuzzle.Player
{
    [System.Serializable]
    public class PlayerGroundChecker
    {
        [Header("Ground Check Settings")]
        [SerializeField] private float sphereRadius = 0.4f;
        [SerializeField] private Vector3 offsetFromCenter = Vector3.down * 0.1f;
        
        public bool IsGrounded(Vector3 playerPosition, float maxDistance, LayerMask groundLayers)
        {
            Vector3 checkPosition = playerPosition + offsetFromCenter;
            
            // Use sphere overlap for reliable ground detection
            Collider[] groundColliders = Physics.OverlapSphere(checkPosition, sphereRadius, groundLayers);
            
            if (groundColliders.Length > 0)
            {
                return true;
            }
            
            // Backup raycast check
            return Physics.Raycast(playerPosition, Vector3.down, maxDistance, groundLayers);
        }
        
        public RaycastHit GetGroundInfo(Vector3 playerPosition, float maxDistance, LayerMask groundLayers)
        {
            Physics.Raycast(playerPosition, Vector3.down, out RaycastHit hit, maxDistance, groundLayers);
            return hit;
        }
        
        public Vector3 GetGroundNormal(Vector3 playerPosition, float maxDistance, LayerMask groundLayers)
        {
            if (Physics.Raycast(playerPosition, Vector3.down, out RaycastHit hit, maxDistance, groundLayers))
            {
                return hit.normal;
            }
            return Vector3.up;
        }
        
        public void DrawDebugGizmos(Vector3 playerPosition, bool isGrounded)
        {
            Vector3 checkPosition = playerPosition + offsetFromCenter;
            
            Gizmos.color = isGrounded ? Color.green : Color.red;
            Gizmos.DrawWireSphere(checkPosition, sphereRadius);
        }
    }
}
